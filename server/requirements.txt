# Convenience file for LOCAL DEVELOPMENT of the server: the same loose version ranges as
# requirements.in. Typical use, from the repository root:
#
#     pip install -e ".[all,dev]"                # the mattstash library from this checkout
#     pip install -r server/requirements.txt -r server/requirements-dev.txt
#
# The Docker images do NOT use this file. They install the hash-pinned server/requirements.lock
# (`pip install --require-hashes`) plus the library from the repository (`pip install .`).
# mattstash is intentionally not listed here: never install it from PyPI for the server, the
# server needs the version in this repository.
-r requirements.in
